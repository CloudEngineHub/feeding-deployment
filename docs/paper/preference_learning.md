\section{Preference Learning}
\label{sec:preference_learning}

\subsection{Problem Formulation}
\label{subsec:problem_formulation}

We consider a single user $u$ interacting with a robot over a sequence of meals $t = 1,\dots,T$. Each user is characterized by fixed physical functional abilities $U^{\mathrm{phys}}_u$ and by a latent, slowly evolving user-specific parameters governing preference formation $U^{\mathrm{pref}}_{u,k(t)}$. The index $k(t) = \lceil t/\Delta \rceil$ partitions the deployment into windows of length $\Delta$, within which $U^{\mathrm{pref}}_{u,k}$ is approximately stationary but may drift over longer horizons. The latent state $U^{\mathrm{pref}}_{u,k}$ encodes stable user-specific tendencies that shape how observable preferences are formed. This state is intrinsic to the user and is not directly observable by the robot.

At each meal $t$, the robot observes an exogenous mealtime context $X_t$, comprising meal composition, dining context (e.g., personal, social, watching TV), and time of day. In addition, the user possesses a latent transient affective state $Y_t$, representing internal factors such as fatigue, hurriedness, or exploratory mood. The affective state $Y_t$ varies across meals and is not directly observable.

For each meal $t$, the user exhibits a multi-dimensional preference vector
\[
P_t = \big(p_t^{(1)}, \dots, p_t^{(M)}\big) \in \mathcal{P},
\]
where $\mathcal{P} := \prod_{i=1}^{M} \mathcal{P}^{(i)}$ is the joint preference space defined as the Cartesian product of per-dimension feasible sets. Each component $p_t^{(i)} \in \mathcal{P}^{(i)}$ corresponds to a distinct preference dimension (e.g., interaction type for bite initiation, robot verbosity level, etc.)

We assume that preference vectors are generated by a shared structural mechanism
\[
P_t \sim f\!\left(
U^{\mathrm{phys}}_u,\;
U^{\mathrm{pref}}_{u,k(t)},\;
X_t,\;
Y_t
\right),
\]
where $f$ defines a joint distribution over $\mathcal{P}$ and captures how physical constraints, user-specific sensitivities, contextual factors, and transient affect interact to produce observable preferences.

\paragraph{Interactive Prediction.}
At the beginning of meal $t$, the robot produces an initial prediction $\hat P_t^{(0)}$. The user may provide a sequence of corrections, each revealing the ground-truth value of a single preference dimension. After $m \ge 0$ corrections, let $S_{t,m} \subseteq \{1,\dots,M\}$ denote the set of revealed indices with $|S_{t,m}| = m$, and let $P_{t,S_{t,m}} = \{p_t^{(i)} : i \in S_{t,m}\}$ denote the corresponding ground-truth values. The unrevealed indices are $\bar S_{t,m} = \{1,\dots,M\} \setminus S_{t,m}$.

The robot conditions on prior interaction history $\mathcal{H}_{t-1}$, which may include past contexts, predictions, corrections, and accepted preference vectors. Based on the observable information, the robot outputs an updated prediction
\[
\hat P_t^{(m)} = g_\theta\!\left(
U^{\mathrm{phys}}_u,\;
X_t,\;
\mathcal{H}_{t-1},\;
S_{t,m},\;
P_{t,S_{t,m}}
\right),
\]
where $g_\theta$ is a parameterized, personalized prediction function. The function $g_\theta$ does not directly observe $U^{\mathrm{pref}}_{u,k(t)}$ or $Y_t$ and must implicitly reason about them through interaction history and partial corrections.

\paragraph{Loss and Objective.}
Because corrected dimensions in $S_{t,m}$ are fixed, prediction reduces to inferring the remaining components $P_{t,\bar S_{t,m}}$. Let $\ell_i(\cdot,\cdot)$ denote a per-dimension loss function and $w_i \ge 0$ optional importance weights. The loss after $m$ corrections is
\[
\mathcal{L}_{t,m}(P_t,\hat P_t^{(m)})
= \sum_{i \in \bar S_{t,m}}
w_i \, \ell_i\!\left(p_t^{(i)}, \hat p_t^{(m,i)}\right).
\]

The learning objective is to minimize expected prediction loss,
\[
\min_{\theta} \;
\mathbb{E}\!\left[
\mathcal{L}_{t,m}(P_t,\hat P_t^{(m)})
\;\middle|\;
U^{\mathrm{phys}}_u,\; X_t,\; \mathcal{H}_{t-1},\;
S_{t,m},\; P_{t,S_{t,m}}
\right],
\]
where the expectation marginalizes over uncertainty induced by the latent user-specific parameters $U^{\mathrm{pref}}_{u,k(t)}$, the latent affective state $Y_t$, and stochasticity in the preference-generation mechanism $f$.

\subsection{Methodology}
\label{subsec:methodology}

We implement $g_\theta$ via a three-component memory architecture. The interaction history $\mathcal{H}_{t-1}$ is represented at two levels of compression: a \emph{long-term memory} $\mathrm{LTM}_t$ that maintains a compressed summary of stable preference patterns accumulated over past meals, and an \emph{episodic memory} $\mathrm{EM}_t$ that retains the full per-meal record without compression. The mealtime context $X_t$ and the corrections revealed so far $(S_{t,m}, P_{t,S_{t,m}})$ are collectively referred to as
the \emph{working memory} $\mathrm{WM}_{t,m}$. The prediction function becomes:
\[
    \hat{P}_t^{(m)} = g_\theta\!\left(U^{\mathrm{phys}}_u,\; \mathrm{LTM}_t,\;
    \mathrm{EM}_{t,m}^{\mathrm{ret}},\; \mathrm{WM}_{t,m}\right),
\]
where $\mathrm{EM}_{t,m}^{\mathrm{ret}} \subseteq \mathrm{EM}_t$ is the subset
of episodes retrieved based on $\mathrm{WM}_{t,m}$. We describe each component
in turn.

% -------------------------------------------------------
\subsubsection{Working Memory}

The working memory is the shortest-lived of the three components, containing only the information from the current meal:
\[
    \mathrm{WM}_{t,m} = \left(X_t,\; S_{t,m},\; P_{t,S_{t,m}}\right),
\]

The working memory is updated after each correction of dimension $j$ of the preference vector from the user:
\[
    S_{t,m+1} = S_{t,m} \cup \{j\}, \qquad P_{t,S_{t,m+1}} = P_{t,S_{t,m}} \cup \{p_t^{(j)}\}.
\]

The working memory is encoded as a structured text string combining context fields and corrected key-value pairs, which serves both as the input to the predictor $g_\theta$ and as the retrieval query for episodic memory.
\begin{quote}
\texttt{meal: pasta; setting: social; time\_of\_day: dinner; speed: slow}
\end{quote}

% -------------------------------------------------------
\subsubsection{Episodic Memory}

The episodic memory $\mathrm{EM}_t = \{h_1, \dots, h_{t-1}\} \subset \mathcal{H}_{t-1}$ stores the complete history of completed meals as individually retrievable episodes, retaining the full per-meal record without compression. Each episode $h_\tau$ encodes the final accepted preference bundle at day $\tau$:
\[
    h_\tau = \bigl(\tau,\; X_\tau,\; P_\tau^*\bigr),
\]
represented as a structured text string of the form
\begin{quote}
\texttt{day=$\tau$; meal=...; setting=...; time\_of\_day=...; preferences: field$_1$=val$_1$ ... field$_M$=val$_M$}.
\end{quote}

\paragraph{Embedding-based retrieval.}
At each prediction step $(t, m)$, we embed the current working memory text $\mathrm{WM}_{t,m}$ and each stored episode $h_\tau$ using a pretrained text embedding function $e: \mathrm{String} \to \mathbb{R}^d$, and retrieve the $K$ episodes with the highest cosine similarity to the working
memory. The retrieved set is denoted $\mathrm{EM}_{t,m}^{\mathrm{ret}} = \{h_{\tau_1}, \dots, h_{\tau_K}\}$. Because the retrieval query evolves with each correction, the retrieved episodes
shift throughout the meal: at $m=0$, retrieval is driven by $X_t$ alone, surfacing episodes with similar meal and setting context, while as corrections accumulate and partially reveal $P_t$, retrieval shifts toward episodes sharing those preference values. This shift is effective because preferences are correlated bundles jointly influenced by the latent affective state $Y_t$: episodes sharing partial preference values are likely to have been generated
under a similar $Y_\tau$, making their full accepted bundles $P_\tau^*$ informative priors for the unrevealed dimensions $P_{t,\bar{S}_{t,m}}$.

% -------------------------------------------------------
\subsubsection{Long-Term Memory}

The long-term memory $\mathrm{LTM}_t$ provides a compressed prior over the stable component of preferences, patterns attributable to $U^{\mathrm{pref}}_{u,k(t)}$ rather than the transient $Y_t$, which the episodic memory alone cannot capture since similarity-based retrieval is context-driven and agnostic to temporal drift. The long-term memory is updated every $\Delta$ meals at window boundaries:
\[
    w(t) = \left\lfloor \frac{t - 1}{\Delta} \right\rfloor, \qquad \mathrm{LTM}_t = \mathrm{LTM}_{w(t)}.
\]
where $\Delta$ is chosen large enough to smooth over one-off episodes driven
by transient $Y_t$, yet small enough to track gradual drift in $U^{\mathrm{pref}}_{u,k(t)}$.

\paragraph{Hierarchical summarization.}
The long-term memory is updated incrementally at each window boundary rather than by re-summarizing the full episode history from scratch:
\[
    \mathrm{LTM}_w = \mathrm{LLM}_{\mathrm{summarize}}\!\left(U^{\mathrm{phys}}_u,\; \mathrm{LTM}_{w-1},\; \{h_\tau\}_{\mathrm{new}}\right),
\]
where $\{h_\tau\}_{\mathrm{new}}$ denotes the episodes collected since the previous update. This incremental scheme ensures that preferences established early in deployment persist through the summary chain even if they do not recur in the most recent window, and that no information is lost to a fixed sliding window. The summary thereby grows in representational richness over time, accumulating increasingly refined conditional patterns linking context $X_t$ to preferred behaviors.

\paragraph{Cold start.}
At the first window boundary ($w = 0$), no episode history is available. The initial long-term memory $\mathrm{LTM}_0$ is generated from $U^{\mathrm{phys}}_u$ alone, providing physically grounded default preferences as a prior for early deployment.

% -------------------------------------------------------
\subsubsection{Baselines}

We compare against baselines that vary the degree of cross-dimension coupling and the model family used for prediction.

\begin{itemize}[leftmargin=*, labelindent=0pt]
    \item \emph{Contextual Bandits}: Each preference dimension is learned independently using a contextual bandit that maps meal context to option values. Corrections in one dimension carry no signal to others.
    \item \emph{Graph Neural Network (GNN)}: A GNN over the $M$ preference dimensions enables joint prediction, modeling cross-dimension dependencies through learned graph structure~\cite{kipf2017semi}.
    \item \emph{LLM Per-Dimension}: An LLM predicts each dimension independently. Unlike our approach, a correction to one dimension does not update predictions for correlated dimensions.
\end{itemize}

We further ablate long-term memory and episodic memory independently within our approach, and remove both together (\emph{No Memory}), to isolate the contribution of each component.

\subsubsection{Evaluation Loop}
\begin{algorithm}[htbp]
\caption{Evaluation Loop at Meal $t$}
\KwIn{$U^{\mathrm{phys}}_u$, $\mathrm{EM}_t$, $X_t$, ground-truth bundle $P_t^*$}
\KwOut{$\hat{P}_t^{(m_t^*)}$}

\BlankLine
\tcp{Long-term memory update}
$w \leftarrow \lfloor (t-1) / \Delta \rfloor$\;
\If{$t \equiv 1 \pmod{\Delta}$}{
    $\mathrm{LTM}_w \leftarrow \mathrm{LLM}_{\mathrm{summarize}}(U^{\mathrm{phys}}_u,\; \mathrm{LTM}_{w-1},\; \{h_\tau\}_{\mathrm{new}})$\;
}

\BlankLine
\tcp{In-meal prediction loop}
$m \leftarrow 0$, $S_{t,0} \leftarrow \emptyset$\;
\Repeat{$\hat{P}_t^{(m)} = P_t^*$ on all $i \in \bar{S}_{t,m}$}{
    $\mathrm{WM}_{t,m} \leftarrow (X_t,\; S_{t,m},\; P_{t, S_{t,m}})$\;
    $\mathrm{EM}_{t,m}^{\mathrm{ret}} \leftarrow \mathrm{TopK}_\tau\;\mathrm{CosSim}(e(\mathrm{WM}_{t,m}),\; e(h_\tau))$\;
    $\hat{P}_t^{(m)} \leftarrow g_\theta(U^{\mathrm{phys}}_u,\; \mathrm{LTM}(t),\; \mathrm{EM}_{t,m}^{\mathrm{ret}},\; \mathrm{WM}_{t,m})$\;
    \If{$\exists\, i \in \bar{S}_{t,m}: \hat{p}_t^{(m,i)} \neq p_t^{*(i)}$}{
        $j \sim \mathrm{Uniform}\!\left(\{i \in \bar{S}_{t,m} : \hat{p}_t^{(m,i)} \neq p_t^{*(i)}\}\right)$\;
        $S_{t,m+1} \leftarrow S_{t,m} \cup \{j\}$, \quad $P_{t,S_{t,m+1}} \leftarrow P_{t,S_{t,m}} \cup \{p_t^{*(j)}\}$\;
        $m \leftarrow m + 1$\;
    }
}
$m_t^* \leftarrow m$\;
$\mathrm{EM}_{t+1} \leftarrow \mathrm{EM}_t \cup \{h_t\}$ \quad where $h_t = (t,\; X_t,\; P_t^*)$\;
\end{algorithm}

The system is evaluated on two metrics:
\begin{itemize}
    \item \emph{Accuracy-after-$m$}: prediction accuracy on the unrevealed
    dimensions $P_{t,\bar{S}_{t,m}}$ as a function of the number of corrections
    $m$ received.
    \item \emph{Mean corrections-to-stop} $\bar{m}^* = \mathbb{E}[m_t^*]$: the
    average number of corrections required before $\hat{P}_t^{(m)}$ agrees with
    $P_t^*$ on all unrevealed dimensions.
\end{itemize}
